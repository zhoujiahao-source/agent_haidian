def pdf_page(c,size,title,figs,lang='zh',sub=''):
    W,H=size; c.setFillColor(HexColor(C['paper'])); c.rect(0,0,W,H,fill=1,stroke=0); c.setFillColor(HexColor(C['ink'])); c.setFont('STSong-Light' if lang=='zh' else 'Helvetica-Bold',22 if W<1000 else 32); c.drawString(22*mm,H-22*mm,title)
    if sub: c.setFont('STSong-Light' if lang=='zh' else 'Helvetica',9 if W<1000 else 13); c.setFillColor(HexColor(C['darkgray'])); c.drawString(22*mm,H-31*mm,sub)
    margin=20*mm; gap=6*mm; top=H-40*mm; bottom=20*mm
    n=len(figs); cols=1 if n==1 else 2; rows=1 if n<=2 else 2; w=(W-2*margin-gap*(cols-1))/cols; h=(top-bottom-gap*(rows-1))/rows
    for i,fp in enumerate(figs):
        col=i%cols; row=i//cols; x=margin+col*(w+gap); y=top-(row+1)*h-row*gap; im=ImageReader(str(fp)); iw,ih=im.getSize(); sc=min(w/iw,h/ih); dw,dh=iw*sc,ih*sc; c.drawImage(im,x+(w-dw)/2,y+(h-dh)/2,dw,dh,preserveAspectRatio=True,mask='auto')
    c.setFillColor(HexColor(C['darkgray'])); c.setFont('STSong-Light' if lang=='zh' else 'Helvetica',7 if W<1000 else 9); c.drawRightString(W-20*mm,9*mm,'Open co-creation proposal · provisional geometry · human final judgment')
