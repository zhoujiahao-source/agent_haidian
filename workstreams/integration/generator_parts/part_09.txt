def draw_pdf_page(c, size, title, fig_paths, lang='zh', subtitle=None, notes=None):
    W,H=size; c.setFillColor(HexColor(COLORS['paper'])); c.rect(0,0,W,H,fill=1,stroke=0)
    c.setFillColor(HexColor(COLORS['ink'])); font='GBSong' if lang=='zh' else 'Helvetica-Bold'; c.setFont(font,24 if W<1000 else 34); c.drawString(24*mm,H-24*mm,title)
    if subtitle:
        c.setFont('GBSong' if lang=='zh' else 'Helvetica',10 if W<1000 else 14); c.setFillColor(HexColor(COLORS['darkgray'])); c.drawString(24*mm,H-32*mm,subtitle)
    top=H-42*mm; margin=22*mm; gap=8*mm
    if len(fig_paths)==1:
        slots=[(margin,24*mm,W-2*margin,top-28*mm)]
    elif len(fig_paths)==2:
        slots=[(margin,24*mm,(W-2*margin-gap)/2,top-28*mm), (margin+(W-2*margin-gap)/2+gap,24*mm,(W-2*margin-gap)/2,top-28*mm)]
    else:
        h=(top-30*mm-gap)/2; w=(W-2*margin-gap)/2; slots=[(margin,top-h,w,h),(margin+w+gap,top-h,w,h),(margin,24*mm,w,h),(margin+w+gap,24*mm,w,h)]
    for fp,slot in zip(fig_paths,slots):
        x,y,w,h=slot; img=ImageReader(str(fp)); iw,ih=img.getSize(); scale=min(w/iw,h/ih); dw,dh=iw*scale,ih*scale; c.drawImage(img,x+(w-dw)/2,y+(h-dh)/2,dw,dh,preserveAspectRatio=True,mask='auto')
    if notes:
        c.setFillColor(HexColor(COLORS['darkgray'])); c.setFont('GBSong' if lang=='zh' else 'Helvetica',7 if W<1000 else 10); c.drawRightString(W-22*mm,10*mm,notes)
