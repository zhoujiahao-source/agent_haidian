def area_sqm(g): return shp_transform(tr.transform,g).area
def length_m(g): return shp_transform(tr.transform,g).length

def feature(fid, layer, geom, source_type='agent_design_proposal', confidence='low', role='design_proposal', **props):
    p={'id':fid,'layer':layer,'source_type':source_type,'confidence':confidence,'geometry_role':role}
    p.update(props)
    if geom.geom_type in ('Polygon','MultiPolygon'): p['area_sqm_declared']=round(area_sqm(geom),3)
    return {'type':'Feature','id':fid,'properties':p,'geometry':mapping(geom)}
def fc(name, feats): return {'type':'FeatureCollection','name':name,'features':feats}
def dumpj(path,obj):
    Path(path).write_text(json.dumps(obj,ensure_ascii=False,indent=2),encoding='utf-8')
